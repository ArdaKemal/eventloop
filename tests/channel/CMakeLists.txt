cmake_minimum_required(VERSION 3.20)
project(eventloop_channel_tests LANGUAGES CXX)

include(CTest)
find_package(eventloop CONFIG REQUIRED)

foreach(name channel channel_allocations)
    add_executable(${name}_test ../${name}.cc)
    target_link_libraries(${name}_test PRIVATE eventloop::eventloop)
    if(MSVC)
        target_compile_options(${name}_test PRIVATE /W4)
    else()
        target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
endforeach()
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # The allocation counter implements global new/delete with malloc/free.
    target_compile_options(channel_allocations_test PRIVATE -Wno-mismatched-new-delete)
endif()
add_test(NAME channel COMMAND channel_test)
add_test(NAME channel_allocations COMMAND channel_allocations_test)
set_tests_properties(channel channel_allocations PROPERTIES TIMEOUT 15)

foreach(mode wrong_loop_send wrong_loop_receive wrong_thread destroy_pending destroy_queued)
    add_test(NAME channel_${mode} COMMAND channel_test ${mode})
    set_tests_properties(channel_${mode} PROPERTIES TIMEOUT 15)
endforeach()
